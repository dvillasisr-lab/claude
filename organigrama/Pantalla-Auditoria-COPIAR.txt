- btnRunA:
    Control: Classic/Button@2.2.0
    Properties:
      Text: =If(varAuditando, "Running...", "Run audit")
      X: =Parent.Width - 400
      Y: =14
      Width: =170
      Height: =36
      Size: =11
      Fill: =Color.White
      HoverFill: =ColorValue("#EEEEF5")
      Color: =ColorValue("#202054")
      FontWeight: =FontWeight.Semibold
      BorderThickness: =0
      RadiusTopLeft: =18
      RadiusTopRight: =18
      RadiusBottomLeft: =18
      RadiusBottomRight: =18
      DisplayMode: =If(varAuditando, DisplayMode.Disabled, DisplayMode.Edit)
      OnSelect: |-
        =Set(varAuditando, true);
        Set(varFiltroA, "");
        ClearCollect(colAudU, UsuariosdeOffice365.SearchUserV2({isSearchTermRequired: false, top: 999}).value);
        ClearCollect(colAudAct, ForAll(Filter(colAudU, AccountEnabled = true), {Value: Id}));
        Refresh(Organigrama);
        ClearCollect(colAudHide, ForAll(Filter(Organigrama, Hide = true), {Value: Lower(Correo)}));
        ClearCollect(colAudit,
          ForAll(Filter(colAudU, AccountEnabled = true) As u,
            With({m: IfError(UsuariosdeOffice365.ManagerV2(u.Id, {'$select': "id,displayName,mail"}), Blank()), svc: Lower(Coalesce(u.Department, "")) = "service account", correo: Lower(Coalesce(u.Mail, u.UserPrincipalName))},
              With({p: Filter(Table(
                  {T: "No manager", C: !svc && IsBlank(m.id)},
                  {T: "Manager is blocked", C: !IsBlank(m.id) && !(m.id in colAudAct.Value)},
                  {T: "No department", C: IsBlank(u.Department)},
                  {T: "Department has no code", C: !svc && !IsBlank(u.Department) && !(" - " in u.Department)},
                  {T: "No job title", C: !svc && IsBlank(u.JobTitle)},
                  {T: "No company", C: !svc && IsBlank(u.CompanyName)},
                  {T: "No office location", C: !svc && IsBlank(u.OfficeLocation)},
                  {T: "Hidden by hand, still active", C: correo in colAudHide.Value},
                  {T: "Possible duplicate", C: CountIf(colAudU, AccountEnabled = true && Lower(Trim(DisplayName)) = Lower(Trim(u.DisplayName))) > 1},
                  {T: "Not a person?", C: !svc && IsBlank(u.GivenName) && IsBlank(u.Surname)},
                  {T: "Service account with manager", C: svc && !IsBlank(m.id)}
                ), C)},
                {Id: u.Id, Nombre: u.DisplayName, Correo: correo, Puesto: u.JobTitle, Depto: u.Department, Empresa: u.CompanyName, Oficina: u.OfficeLocation, Jefe: m.displayName, Problemas: Concat(p, T, ", "), N: CountRows(p)}))));
        ClearCollect(colAudRes, {T: "All issues", N: CountRows(Filter(colAudit, N > 0))});
        Collect(colAudRes, Sort(Filter(ForAll(["No manager", "Manager is blocked", "No department", "Department has no code", "No job title", "No company", "No office location", "Hidden by hand, still active", "Possible duplicate", "Not a person?", "Service account with manager"] As t, {T: t.Value, N: CountIf(colAudit, t.Value in Problemas)}), N > 0), N, SortOrder.Descending));
        Set(varAuditando, false)
